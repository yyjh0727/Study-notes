### Singleton 单例(饿汉式,懒汉式)
- 类只能创建一个对象
- 任何位置获得这个类的实例都是同一个

实现
- 构造方法私有化，封装，屏蔽了访问方法
- 提供一个类接口，获得唯一实例(对象)

```c++
//singleton 单例
#include <iostream>

using namespace std;

class Singleton
{
private:
    Singleton(/* args */) {}    //构造方法私有化
    
public:
    ~Singleton() {}
    //static Singleton instance; 饿汉式 声明
    //开放一个静态的接口
    static Singleton *getInstanc()
    {
        //懒汉式 直到调用才创建对象
        static Singleton instance;
        return &instance;
    }
};
//Singleton Singlrton :: instance; 饿汉式 定义
int main(int argc, char const *argv[])
{
    Singleton *s1 = Singleton::getInstanc();
    Singleton *s2 = Singleton::getInstanc();
    Singleton *s3 = Singleton::getInstanc();
    cout << s1 <<endl;
    cout << s2 <<endl;
    cout << s3 <<endl;
    return 0;
}

```

## 工厂模式
```C++
class Doc
{
protected:
    string name;

public:
    Doc(string name) : name(name) {}
    ~Doc() {}
    void open()
    {
        cout << "OPEN:" << name << endl;
    }
    virtual void edit() = 0;
};

class Word : public Doc
{
private:
    /* data */
public:
    Word(string name) : Doc("name") {}
    ~Word() {}
    void edit() override
    {
        cout << "EDIT Word: " << name << endl;
    }
};

class Ppt : public Doc
{
private:
    /* data */
public:
    Ppt(string name) : Doc("name") {}
    ~Ppt() {}
    void edit() override
    {
        cout << "EDIT Ppt: " << name << endl;
    }
};

class Excel : public Doc
{
private:
    /* data */
public:
    Excel(string name) : Doc("name") {}
    ~Excel() {}
    void edit() override
    {
        cout << "EDIT Excel: " << name << endl;
    }
};

enum DocType
{
    WORD, PPT, EXCEL
};

class WPS
{
public:
    WPS() {}
    ~WPS() {}
    static Doc *create(string name, DocType type)
    {
        Doc *doc;
        switch (type)
        {
        case WORD:
            doc = new Word(name);
            break;
        case PPT:
            doc = new Ppt(name);
        case EXCEL:
            doc = new Excel(name);
        default:
            break;
        }
    }
};
```
